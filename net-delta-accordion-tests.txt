/* eslint-disable @typescript-eslint/no-unused-vars */
import { testBonds } from '@bondsData/test-bonds'
import { test } from '@fixtures'
import { softExpect } from '@helpers/assertions'
import { raiseSwitchRfqFromRabbitMq, raiseButterflyRfqFromRabbitMq, rejectRmqRfq } from '@rabbitmq'
import { generateRmqBreakEvenSwitchData, generateRmqButterflyData } from 'src/factories/rabbitmq.factory'

test('Should verify Net Delta label, calculation, and accordion default state on Switch RFQ @critical @rfq-flow', async ({
    darwinLauncher,
    request,
    darwinRfq,
    darwinSettings,
}, testInfo) => {
    testInfo.annotations.push({ type: 'test_key', description: 'DWB-2521' })
    let rfqIdNumber: number

    await test.step('I configure a high Net Delta limit in User Settings', async () => {
        await darwinSettings.openNetDeltaLimitSettings()
        await darwinSettings.netDeltaLimitPage.setSwitchNetDeltaLimit('10')
        await darwinSettings.saveChanges()
    })

    await test.step('I raise a Switch RFQ via RMQ', async () => {
        const rfqData = generateRmqBreakEvenSwitchData(
            'Buy',
            'Sell',
            2000000,
            1000000,
        )
        rfqIdNumber = rfqData.rfqIdNumber
        await raiseSwitchRfqFromRabbitMq(request, rfqData)
    })

    await test.step('I verify the pricing section accordion is collapsed by default', async () => {
        await darwinRfq.verifyPricingSectionCollapsed()
    })

    await test.step('I expand the pricing section accordion manually', async () => {
        await darwinRfq.expandPricingSection()
        await darwinRfq.verifyPricingSectionExpanded()
    })

    await test.step('I verify Net Delta label is present with a computed value', async () => {
        await softExpect(darwinRfq.netDeltaValueContainer).toBeVisible()
        const netDelta = await darwinRfq.netDeltaValueContainer.textContent()
        softExpect(netDelta).toBeTruthy()
    })

    await test.step('I verify Net Delta is NOT highlighted when within limit', async () => {
        await darwinRfq.verifyNetDeltaNoHighlight()
    })

    await test.step('I reject the RFQ to clean up', async () => {
        await rejectRmqRfq(request, rfqIdNumber)
        await darwinRfq.waitForRfqStackToBeEmpty()
    })
})
